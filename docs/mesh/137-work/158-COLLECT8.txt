#: Provider error text that means *this route does not exist any more*, as
#: opposed to "the request was bad". Measured 2026-10-01 on the authority:
#: Fireworks answered `HTTP 404 {"code":"NOT_FOUND","message":"Model not found,
#: inaccessible, and/or not deployed"}` for
#: `fireworks/accounts/fireworks/models/deepseek-v4-pro-0813` — a model the live
#: catalogue (`GET /v1/models`) still lists — and because 404 was not in the
#: fallback-eligible set, the gateway stopped on the first candidate and every
#: step of four standing work orders died on the same call.
_MODEL_UNAVAILABLE_MARKERS = (
    "model not found",
    "not deployed",
    "inaccessible",
    "model_not_found",
    "invalid_model",
    "unknown model",
    "does not exist",
    "no such model",
)


def _looks_like_model_unavailable(payload: Any) -> bool:
    """True when a provider error body names a model it cannot serve."""
    error = payload.get("error") if isinstance(payload, dict) else None
    if isinstance(error, dict):
        code = str(error.get("code") or "").strip().lower()
        message = str(error.get("message") or "").strip().lower()
    elif isinstance(error, str):
        code, message = "", error.strip().lower()
    else:
        code, message = "", ""
    if code in {"model_not_found", "invalid_model", "not_found", "model_unavailable"}:
        return True
    blob = f"{code} {message}"
    return any(marker in blob for marker in _MODEL_UNAVAILABLE_MARKERS)


def _http_reason_code(exc: httpx.HTTPStatusError) -> str:
    status = exc.response.status_code if exc.response is not None else 0
    if status in (400, 404, 410) and exc.response is not None:
        try:
            payload = exc.response.json()
            error = payload.get("error") if isinstance(payload, dict) else None
            if isinstance(error, dict):
                code = str(error.get("code") or "").lower()
                message = str(error.get("message") or "").lower()
                if code == "unsupported_parameter" or "unsupported parameter" in message:
                    return "http_400_unsupported_parameter"
            if _looks_like_model_unavailable(payload):
                # One meaning -- "the pinned route is dead" -- whatever status the
                # provider chose to report it with (400 model_not_found, 404
                # NOT_FOUND, 410 gone). Both codes are fallback-eligible; the
                # name keeps the provider's own status so the log stays honest.
                if status == 400:
                    return "http_400_model_unavailable"
                return "http_404_model_unavailable"
        except Exception:
            pass
    return f"http_{status or 'unknown'}"


def _should_try_fallback(reason_code: str) -> bool:
    if reason_code.startswith("vision_"):
        return True
    if reason_code in _BAD_SUCCESS_FALLBACK_REASONS:
        return True
    if reason_code in {"http_400_unsupported_parameter", "http_400_model_unavailable"}:
        return True
    if reason_code.startswith("http_"):
        match = re.match(r"http_(\d{3})", reason_code)
        if not match:
            return False
        status = int(match.group(1))
        # 402 = payment required / out of credits (OpenRouter balance exhausted)
        # 404/410 = the routed URL or model is not there any more: a dead route
        # is exactly what the escalation chain exists for, so it must be
        # retried on the next candidate rather than ending the request.
        return status in {402, 404, 408, 409, 410, 425, 429} or status >= 500
    return False
