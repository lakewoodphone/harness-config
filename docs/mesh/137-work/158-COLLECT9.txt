# ── Upstream resolution ────────────────────────────────────

# NOTE: ``_GOOGLE_MODEL_RENAME`` and ``billing_model_id`` live at the top of
# this module so every cost path (including the usage aggregation helpers)
# can resolve the billed model name without depending on definition order.


def _resolve_upstream(
    settings: Settings,
    *,
    provider: str,
    provider_model_id: str,
    canonical_id: str,
) -> tuple[str, str, str]:
    """Return (base_url, api_key, upstream_model) for a routed model.

    Raises ``RuntimeError`` if no reachable upstream exists.
    """
    keys = settings.provider_api_keys()

    # Native OpenAI-compatible providers.
    if provider in _OPENAI_COMPAT_UPSTREAM:
        base_url, key_attr = _OPENAI_COMPAT_UPSTREAM[provider]
        api_key = str(getattr(settings, key_attr, None) or keys.get(provider) or "")
        if api_key:
            if provider == "google":
                # Map stale catalog names to live Google models (see
                # _GOOGLE_MODEL_RENAME) so auto-routed picks don't 404.
                provider_model_id = _GOOGLE_MODEL_RENAME.get(
                    provider_model_id, provider_model_id
                )
            return base_url, api_key, provider_model_id

    # Anthropic / Google via OpenRouter translation — DISABLED June 2026.
    # Both now have native OpenAI-compatible endpoints in _OPENAI_COMPAT_UPSTREAM.
    # _OPENROUTER_TRANSLATABLE_PROVIDERS is empty; this branch is dead code.

    # Last resort: if the canonical id is already an OpenRouter slug — DISABLED.
    # OpenRouter removed June 2026 (too expensive).

    raise RuntimeError(
        f"No reachable upstream for {canonical_id} (provider={provider})"
    )
