def _estimate_cost(
    canonical_id: str,
    usage: dict[str, Any],
    *,
    cache_aware_enabled: bool = False,
    priced_model_id: str | None = None,
) -> float:
    """Estimate the USD cost of one call.

    ``priced_model_id`` is the model name that will be recorded in
    ``model_usage`` and billed upstream. When it differs from ``canonical_id``
    (Google catalog renames — see :func:`billing_model_id`) the price must come
    from the billed name, or the ledger prices a model we never called.
    """
    price_id = canonical_id
    if priced_model_id:
        price_id = billing_model_id(priced_model_id)
        # Some providers key the catalog with the provider prefix (the router's
        # canonical ids) and some without (the raw upstream model names). Prefer
        # whichever form the price table knows, so a bare upstream name like
        # ``gemini-3.5-flash`` cannot silently price at $0 (or, worse, fall
        # through to the $5/M unknown-model default) while
        # ``google/gemini-3.5-flash`` is rated at $9/M output.
        if price_id not in sr._CAPABILITY_DB:
            candidates = [price_id]
            # Provider hint from whichever id the caller actually gave us.
            for source in (priced_model_id, canonical_id):
                hint = str(source or "").partition("/")[0]
                if hint and hint != price_id:
                    candidates.append(f"{hint}/{price_id}")
            caller_hint = str(priced_model_id or canonical_id or "").partition("/")[0]
            if caller_hint:
                bare = price_id.rsplit("/", 1)[-1]
                candidates.append(f"{caller_hint}/{bare}")
            for candidate in candidates:
                if candidate in sr._CAPABILITY_DB:
                    price_id = candidate
                    break
    info = sr._CAPABILITY_DB.get(price_id, {})
    pin = float(info.get("cost_per_mtok_input", 0) or 0)
    pout = float(info.get("cost_per_mtok_output", 0) or 0)
    pcache = float(info.get("cost_per_mtok_cache_read", pin) or pin)
    prompt = float(usage.get("prompt_tokens", 0) or 0)
    completion = float(usage.get("completion_tokens", 0) or 0)
    cached = float(extract_cached_tokens(usage)) if cache_aware_enabled else 0.0
    cached = min(max(cached, 0.0), prompt)
    uncached = max(prompt - cached, 0.0)

    # If _CAPABILITY_DB returned zero pricing (e.g. Copilot models not in DB),
    # fall back to pricing.py which has per-token rates for all providers.
    if pin == 0 and pout == 0 and (prompt > 0 or completion > 0):
        try:
            from app.pricing import estimate_cost as pricing_estimate

            provider = price_id.split("/", 1)[0] if "/" in price_id else ""
            model = price_id.split("/", 1)[1] if "/" in price_id else price_id
            est = pricing_estimate(
                provider=provider,
                model=model,
                prompt_tokens=int(prompt),
                completion_tokens=int(completion),
            )
            if est.estimated_cost_usd > 0:
                return est.estimated_cost_usd
        except Exception:
            pass

    return (uncached * pin + cached * pcache + completion * pout) / 1_000_000.0


def _billed_model_for(prepared: dict[str, Any]) -> str:
    """The single name that is both priced and recorded for a prepared request.

    ``_record`` writes this exact string into ``model_usage.model`` and
    ``_estimate_cost`` prices with it, so a recorded row can always be
    recomputed from its own model name.
    """
    upstream = str((prepared or {}).get("upstream_model") or "").strip()
    if upstream:
        return upstream
    return str((prepared or {}).get("canonical_id") or "")
