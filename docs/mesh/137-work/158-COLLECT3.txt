#!/usr/bin/env python3
"""ai-gateway-health — does any model we actually call or configure look broken?

WHY THIS EXISTS
2026-09-17: `model_tier1`, `model_tier2` and `model_tier3` all pointed at the literal
`deepseek/deepseek-v4-flash`, which failed **655 of 890 calls in 7 days — every failure `empty_response`**
— and stopped being called entirely on 2026-09-15T21:26Z. Nothing raised it. It was found only because an
unrelated owner-queue row was investigated and the numbers were read by hand.

An `empty_response` is the worst failure mode to leave unmonitored: a caller sees a model with nothing to
say, not an error, so a fallback answers and the output stays *plausible* while the configured model is
broken. The standing instruction was written for exactly this shape: *"ai models change very often, can
never be hardcoded... we need robust easy flows for that."*

WHAT IT CHECKS
  1. FAILING MODELS   a model with enough calls to judge whose failure rate exceeds a threshold
  2. SILENT MODELS    a model with ZERO calls in the window while being configured for a tier — the
                      "configured but not actually serving" case, which lookups alone cannot see
  3. TIER COLLAPSE    tier1, tier2 and tier3 resolving to the SAME model, which removes the redundancy
                      that would otherwise surface a bad one

EXIT  0 healthy · 3 a CONDITION found (external, not ours to fix) · 1 a FAULT found · 2 could not measure
Tri-state, the same shape as `text-ledger-audit.py`: `cron-wrap.sh` records the code, so runs.tsv
carries the distinction instead of every finding looking like the same red. A FAULT ALWAYS WINS over a
condition — rc=3 is returned only when EVERY finding is an external provider condition (a provider
account out of balance, or a provider that no longer deploys the id), and rc=1 the moment anything is
ours to fix. An UNMEASURED check is reported as a failure, never as health — an empty result is not evidence.

USAGE
  ai-gateway-health.py                 # human summary
  ai-gateway-health.py --json          # machine
  ai-gateway-health.py --days 7        # widen the window
  ai-gateway-health.py --min-calls 20  # how many calls before a rate is judgeable
  ai-gateway-health.py --self-test     # run the built-in negative control and exit
"""

# A rate below this is normal provider noise. MEASURED context for the threshold: the model this was
# written for sat at 74%, the healthy DeepInfra route at 0%, and the busiest normal model at 0%.
FAIL_RATE_LIMIT = 0.20

# ── ONE FAILURE IS AN EVENT, NOT A PATTERN ──────────────────────────────────────────────────────────────────────────────────────────────
# MEASURED 2026-10-05: this job was red for 138 consecutive runs because a lane with 3 calls in the
# recent window and ONE `finish_reason_length` truncation produced recent_rate=33% > FAIL_RATE_LIMIT
# and was graded a FAULT (rc=1). The same truncation hit four unrelated lanes in twelve hours
# (gemini-3.5-flash, gemini-3.1-pro-preview, deepseek-v4-pro, gemini-2.5-pro) and each lane's next
# call succeeded. `finish_reason_length` is the CLIENT's max_tokens cap being hit, which the gateway
# itself accepts on the final attempt ("the client's own max_tokens cap was hit", ai_gateway.py) --
# response-shape noise, not a broken model. Two is the smallest count that can be a pattern; a
# fully-failing recent sample still fires, because that is >= 2 failures.
MIN_RECENT_FAILURES = 2

# ── CONDITION vs FAULT ──────────────────────────────────────────────────────────────────────────────
# A model that fails because a PROVIDER ACCOUNT is out of balance, or because the provider no longer
# deploys the id, is not broken: nothing in this repo can fix it, and the finding is already escalated
# (the 2026-10-03 Deep Infra post-mortem and standing-order tasks #27454/#27456). Reporting that as a
# FAULT (rc=1) every 30 minutes is what made this job red 104 runs in a row and hid any real fault in
# the same colour — the exact failure mode `text-ledger-audit.py` solved with exit 3. MEASURED
# 2026-10-05: all four "failing-now" lanes were 402 "You need positive balance" / 404 "Model not found,
# inaccessible, and/or not deployed", and the matching Deep Infra payment failure was already known.
#
# This repo's word for it is exit 3 = "condition, not fault" (order-feed-refresh, mesh-hygiene,
# serve-phone all use it; `sources/cron-failures.py` reads a known rc as a signal, not a fault).
CONDITION_EXIT = 3

#: Provider-side failure text meaning "the ACCOUNT or DEPLOYMENT is the problem", not "the model is
#: broken". Matched case-insensitively against `model_usage.error_message`.
CONDITION_ERROR_PATTERNS = (
    # `(?<!\d)402(?!\d)` and NOT `\b402\b`: Deep Infra records `http_402: {...}`, and '_' is a word
    # character, so there is no \b between '_' and '4' and the bare prefix would not match. MEASURED
    # 2026-10-05 01:30: a lane whose only new failure was `http_402:` read as a FAULT and flipped the
    # whole run from rc=3 to rc=1 for one tick.
    re.compile(r"(?<!\d)402(?!\d)|positive balance|insufficient balance|insufficient credit|"
               r"quota exceeded", re.I),
    re.compile(r"model not found|not_found_error|does not exist or you do not have access|"
               r"not deployed|no such model|invalid model id", re.I),
    re.compile(r"temporary failure in name resolution", re.I),
)


def _is_condition_error(msg: str) -> bool:
    """True when this failure text names a provider ACCOUNT/DEPLOYMENT condition, not a broken model."""
    if not msg:
        return False
    return any(p.search(msg) for p in CONDITION_ERROR_PATTERNS)


def _connect(db: str) -> sqlite3.Connection:
    con = sqlite3.connect("file:%s?mode=ro" % db, uri=True)
    con.execute("PRAGMA query_only=1")
    return con
